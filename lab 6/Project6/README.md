#include <iostream>
#include <clocale>
using namespace std;

// ===== 1. Факториал =====
long long factorial(int n) {
    if (n <= 1) return 1;
    return n * factorial(n - 1);
}

// ===== 2. Числа Фибоначчи =====
long long fib(int n) {
    if (n <= 1) return n;
    return fib(n - 1) + fib(n - 2);
}

// ===== 3. Максимум в массиве =====
int findMax(int arr[], int n) {
    if (n == 1) return arr[0];
    int maxOfRest = findMax(arr, n - 1);
    return (arr[n - 1] > maxOfRest) ? arr[n - 1] : maxOfRest;
}

// ===== 4. Ханойские башни =====
long long hanoiMoves = 0;
void hanoi(int n, char from, char to, char aux) {
    if (n == 1) {
        cout << "Переместить диск 1 с " << from << " на " << to << "\n";
        ++hanoiMoves;
        return;
    }
    hanoi(n - 1, from, aux, to);
    cout << "Переместить диск " << n << " с " << from << " на " << to << "\n";
    ++hanoiMoves;
    hanoi(n - 1, aux, to, from);
}

#include <iostream>
#include <cstdlib>
#include <chrono>
#include <windows.h>
using namespace std;
using namespace chrono;

struct Stats {
    long long comparisons = 0; // сравнения
    long long swaps = 0;       // обмены
};

// ===== 1. Пузырьковая =====
void bubbleSort(int arr[], int n, Stats& st) {
    for (int i = 0; i < n - 1; ++i) {
        bool swapped = false;
        for (int j = 0; j < n - i - 1; ++j) {
            ++st.comparisons;
            if (arr[j] > arr[j + 1]) {
                swap(arr[j], arr[j + 1]);
                ++st.swaps;
                swapped = true;
            }
        }
        if (!swapped) break; // оптимизация
    }
}

// ===== 2. Вставками =====
void insertionSort(int arr[], int n, Stats& st) {
    for (int i = 1; i < n; ++i) {
        int key = arr[i];
        int j = i - 1;
        while (j >= 0) {
            ++st.comparisons;
            if (arr[j] > key) {
                arr[j + 1] = arr[j];
                ++st.swaps; // считаем сдвиг как перемещение
                --j;
            }
            else break;
        }
        arr[j + 1] = key;
    }
}

// ===== 3. Выбором =====
void selectionSort(int arr[], int n, Stats& st) {
    for (int i = 0; i < n - 1; ++i) {
        int minIndex = i;
        for (int j = i + 1; j < n; ++j) {
            ++st.comparisons;
            if (arr[j] < arr[minIndex]) minIndex = j;
        }
        if (minIndex != i) {
            swap(arr[i], arr[minIndex]);
            ++st.swaps;
        }
    }
}

// Вспомогательная функция для запуска и вывода статистики
void runTest(const char* name, void (*sortFunc)(int[], int, Stats&), int original[], int n) {
    int* copy = new int[n];
    for (int i = 0; i < n; ++i) copy[i] = original[i];

    Stats st;
    auto start = high_resolution_clock::now();
    sortFunc(copy, n, st);
    auto end = high_resolution_clock::now();
    long long us = duration_cast<microseconds>(end - start).count();

    cout << name
        << " | Сравнений: " << st.comparisons
        << " | Обменов: " << st.swaps
        << " | Время: " << us << " мкс\n";

    delete[] copy;
}

int main() {
    SetConsoleOutputCP(CP_UTF8);
    setlocale(LC_ALL, "Russian");

    const int N = 1000; // размер массива
    int* arr = new int[N];
    srand(42); // фиксируем генератор случайных чисел

    // Заполняем случайными числами
    for (int i = 0; i < N; ++i) arr[i] = rand() % N;

    cout << "Размер массива N = " << N << " (случайные числа)\n";
    cout << "----------------------------------------------------------\n";

    runTest("Пузырьковая", bubbleSort, arr, N);
    runTest("Вставками  ", insertionSort, arr, N);
    runTest("Выбором    ", selectionSort, arr, N);

    delete[] arr;
    return 0;
}


Контрольные вопросы
1. Что такое рекурсия? Из каких компонентов состоит рекурсивная функция?
Рекурсия — это метод программирования, при котором функция вызывает саму себя для решения подзадач меньшего размера. Рекурсивная функция обязательно состоит из двух компонентов:

Базовый случай (условие остановки) — при достижении этого условия функция прекращает вызывать себя и возвращает конкретное значение.

Шаг рекурсии — вызов функции с измененными (обычно меньшими) параметрами, приближающими выполнение к базовому случаю.

2. Что такое базовый случай рекурсии? Почему он важен?
Базовый случай — это условие, при котором рекурсия останавливается (например, if (n == 0) return 1;). Он критически важен, потому что без него функция будет вызывать себя бесконечно. Это приведет к переполнению стека вызовов (Stack Overflow) и аварийному завершению программы.

3. Чем отличается прямая рекурсия от косвенной?

Прямая рекурсия: функция вызывает саму себя напрямую (например, функция A вызывает A).

Косвенная рекурсия: функция вызывает другую функцию, которая в итоге приводит к вызову первой (например, функция A вызывает B, а B вызывает A).

4. Что такое хвостовая рекурсия? В чём её преимущество?
Хвостовая рекурсия — это вид рекурсии, при котором рекурсивный вызов является самой последней операцией в функции (после него нет никаких математических действий с результатом).
Преимущество: умные компиляторы умеют оптимизировать хвостовую рекурсию, превращая её в обычный цикл. Это позволяет избежать переполнения стека и экономит память.

5. Какова временная сложность рекурсивного вычисления чисел Фибоначчи? Почему?
Сложность составляет O(2^n) (экспоненциальная). Это происходит потому, что при вычислении fib(n) функция вызывает себя дважды: для fib(n-1) и fib(n-2). В результате дерево вызовов ветвится, и одни и те же значения вычисляются множество раз.

6. Как можно оптимизировать рекурсивное вычисление Фибоначчи?
Есть два основных способа:

Мемоизация — сохранение уже вычисленных значений в массиве или хэш-таблице. Если значение уже считалось, оно берется из памяти, а не вычисляется заново (сложность падает до O(n)).

Итеративный подход — использование простого цикла и хранения только двух последних чисел (сложность O(n) по времени и O(1) по памяти).

7. В чём отличие рекурсивной реализации от итеративной?

Рекурсивная реализация обычно короче и читается ближе к математическому определению задачи, но она требует дополнительной памяти под стек вызовов (O(n)) и работает медленнее из-за накладных расходов на вызовы функций.

Итеративная реализация (циклы) использует фиксированную память O(1), работает быстрее, но код может выглядеть более громоздко.

8. Какова сложность пузырьковой сортировки? Когда она становится O(n)?
В среднем и худшем случае сложность O(n^2). Сложность становится O(n) (линейной), если массив уже отсортирован. Это достигается благодаря оптимизации: если за очередной проход не произошло ни одного обмена, значит массив отсортирован, и цикл прерывается досрочно.

9. Чем сортировка вставками отличается от сортировки выбором?

Вставками: берет текущий элемент и «вставляет» его на нужное место в уже отсортированную часть массива, сдвигая остальные элементы вправо.

Выбором: ищет минимальный элемент во всей неотсортированной части и меняет его местами с первым элементом этой части. (Вставками делает много сдвигов, а выбором — мало обменов).

10. Какая из простых сортировок является устойчивой? Что это значит?
Устойчивыми являются пузырьковая сортировка и сортировка вставками. Устойчивость означает, что если в массиве есть два одинаковых элемента, то после сортировки они сохранят свой исходный порядок относительно друг друга. (Сортировка выбором — неустойчива).

11. Какая из простых сортировок работает лучше на почти отсортированных данных?
Лучше всего работает сортировка вставками. На почти отсортированном массиве она делает очень мало сдвигов и ее время выполнения приближается к O(n), в то время как сортировка выбором все равно сделает O(n^2) сравнений.

12. Как реализовать пузырьковую сортировку с оптимизацией?
Нужно ввести логический флаг (например, swapped = false) во внешний цикл. Если во внутреннем цикле произошел обмен, флаг становится true. Если после завершения внутреннего цикла флаг остался false — значит, обменов не было, массив уже отсортирован, и нужно досрочно прервать внешний цикл (break).

13. Что такое рекурсивный поиск с возвратом (backtracking)?
Это метод перебора вариантов решения задачи, при котором мы делаем шаг вперед, а если понимаем, что зашли в тупик — возвращаемся на шаг назад и пробуем другой вариант. Это как прохождение лабиринта «наугад» с возможностью отступать.

Сложность алгоритма — O(2^n) (экспоненциальная). Количество ходов, необходимых для перемещения n дисков, вычисляется по формуле 2^n - 1. (Например, для 5 дисков нужно 2^5 - 1 = 31 ход, что и подтвердила наша программа).
